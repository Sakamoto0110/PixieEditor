cmake_minimum_required(VERSION 3.25)
project(PixieEditorNative LANGUAGES C CXX)

if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "PixieEditor native qualification requires MSVC on Windows x64.")
endif()

set(IMGUI_WCHAR32 ON CACHE BOOL "" FORCE)

if(PIXIE_NATIVE_TARGET STREQUAL "cimgui")
    find_package(ZLIB REQUIRED)
    set(IMGUI_FREETYPE ON CACHE BOOL "" FORCE)
    set(CIMGUI_VARGS0 ON)
    add_subdirectory("${PIXIE_NATIVE_SOURCE}" upstream)
    target_sources(cimgui PRIVATE freetype-exports.cpp)
    target_link_options(cimgui PRIVATE /MAP)
elseif(PIXIE_NATIVE_TARGET STREQUAL "base" OR PIXIE_NATIVE_TARGET STREQUAL "glfw")
    if(PIXIE_NATIVE_TARGET STREQUAL "base")
        set(BACKENDS_BASE_ONLY ON CACHE BOOL "" FORCE)
    else()
        set(BACKENDS_GLFW_ONLY ON CACHE BOOL "" FORCE)
    endif()
    add_subdirectory("${PIXIE_NATIVE_SOURCE}" upstream)
else()
    message(FATAL_ERROR "Choose cimgui, base or glfw for PIXIE_NATIVE_TARGET.")
endif()
